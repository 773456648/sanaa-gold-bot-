// بكلمك حر: هذا الكود هو المحرك الحقيقي اللي يفك التشفير ويسحب النت const net = require('net'); const WebSocket = require('ws'); const http = require('http'); const express = require('express'); const app = express(); const port = process.env.PORT || 3000; // المعرف الخاص بك const uuid = '4ba66ace-7517-46c7-a22c-23f7729b5d5a'; // 1. واجهة الموقع لتوليد الرابط app.get('/', (req, res) => { const host = req.get('host'); const vlessLink = `vless://${uuid}@${host}:443?encryption=none&security=tls&type=ws&host=${host}&path=%2f#Sanaa_Gold_Net`; res.send(`

Sanaa Gold - Internet Active 🌐

المحرك الآن جاهز لسحب الإنترنت وتوجيهه لجوالك.

${vlessLink}

انسخ الرابط، احذف القديم، استورد هذا، ومبروك النت!

`); }); const server = http.createServer(app); const wss = new WebSocket.Server({ server }); // 2. المحرك الحقيقي لسحب الإنترنت (هنا الزبدة عشان يانتر) wss.on('connection', (ws) => { ws.once('message', (msg) => { const [VERSION] = msg; const id = msg.slice(1, 17); // التحقق من أن الاتصال قادم من جوالك if (!id.every((v, i) => v === parseInt(uuid.replace(/-/g, '').slice(i * 2, i * 2 + 2), 16))) { return; } let i = msg.readUInt8(17) + 19; const port = msg.readUInt16BE(i); i += 2; const ATYP = msg.readUInt8(i); i += 1; // معرفة الموقع أو اللعبة اللي تشتي تفتحها let host = ''; if (ATYP === 1) { host = msg.slice(i, i + 4).join('.'); i += 4; } else if (ATYP === 2) { const len = msg.readUInt8(i); i += 1; host = msg.slice(i, i + len).toString('utf8'); i += len; } else if (ATYP === 3) { host = msg.slice(i, i + 16).reduce((s, b, j, a) => (j % 2 ? s.concat(a.slice(j - 1, j + 1).toString('hex')) : s), []).join(':'); i += 16; } // توجيه البيانات للإنترنت الحقيقي ws.send(new Uint8Array([msg[0], 0])); const duplex = WebSocket.createWebSocketStream(ws); const netSocket = net.connect({ host: host, port: port }, () => { duplex.pipe(netSocket).pipe(duplex); }); netSocket.on('error', () => { ws.close(); }); }); }); server.listen(port, () => { console.log(`VPN Core is running on port ${port}`); });
